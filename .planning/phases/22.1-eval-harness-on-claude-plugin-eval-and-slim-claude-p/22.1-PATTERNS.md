function slimRun(runSlim: string, args: readonly string[]): Promise<number> {
  const out = args[args.indexOf('--out') + 1] ?? '';

  return new Promise((done) => {
    const child = spawn(process.execPath, [runSlim, ...args], { stdio: 'inherit' });

}
